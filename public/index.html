const express = require('express'); const cors = require('cors'); const helmet = require('helmet'); const rateLimit = require('express-rate-limit'); const path = require('path'); const db = require('./db'); const config = require('./config'); const { requireAuth, signToken, getUserFromLogin, ensureDefaultAdmin } = require('./auth'); const app = express(); app.use(helmet({ contentSecurityPolicy: false })); app.use(cors()); app.use(express.json({ limit: '2mb' })); app.use(express.urlencoded({ extended: true })); app.use(rateLimit({ windowMs: 60 * 1000, max: 120 })); const publicDir = path.join(__dirname, '..', 'public'); app.use(express.static(publicDir)); function logAudit(actor, action, entityType, entityId, details) { db.prepare(`INSERT INTO audit_logs (actor, action, entity_type, entity_id, details) VALUES (?, ?, ?, ?, ?)`).run( actor || 'system', action, entityType || null, entityId || null, details || '' ); } function getDashboardData() { const totalTickets = db.prepare('SELECT COUNT(*) as value FROM tickets').get().value; const openTickets = db.prepare("SELECT COUNT(*) as value FROM tickets WHERE status = 'open'").get().value; const closedTickets = db.prepare("SELECT COUNT(*) as value FROM tickets WHERE status = 'closed'").get().value; const pendingTickets = db.prepare("SELECT COUNT(*) as value FROM tickets WHERE status = 'pending'").get().value; const inProgressTickets = db.prepare("SELECT COUNT(*) as value FROM tickets WHERE status = 'in_progress'").get().value; const activePanels = db.prepare("SELECT COUNT(*) as value FROM panels WHERE status = 'published'").get().value; const totalPanels = db.prepare('SELECT COUNT(*) as value FROM panels').get().value; const avgResolution = db.prepare(` SELECT ROUND(AVG(CAST((strftime('%s', closed_at) - strftime('%s', created_at)) AS REAL) / 3600), 2) as value FROM tickets WHERE status = 'closed' AND closed_at IS NOT NULL `).get().value || 0; const recentTickets = db.prepare(` SELECT * FROM tickets ORDER BY created_at DESC LIMIT 6 `).all(); const recentActivity = db.prepare(` SELECT * FROM audit_logs ORDER BY created_at DESC LIMIT 10 `).all(); const botState = db.prepare('SELECT * FROM bot_status ORDER BY id DESC LIMIT 1').get(); return { totals: { tickets: totalTickets, open: openTickets, closed: closedTickets, pending: pendingTickets, inProgress: inProgressTickets, panels: totalPanels, activePanels, averageResolutionHours: avgResolution, botOnline: Boolean(botState?.is_online), lastSync: botState?.last_sync || null, status: botState?.error || 'Sem erro registrado', }, recentTickets, recentActivity, }; } app.get('/api/health', (req, res) => { res.json({ ok: true, app: config.appName, uptime: process.uptime() }); }); app.post('/api/auth/login', async (req, res) => { const { email, password } = req.body || {}; if (!email || !password) { return res.status(400).json({ ok: false, message: 'E-mail e senha são obrigatórios.' }); } const user = await getUserFromLogin(email, password); if (!user) { return res.status(401).json({ ok: false, message: 'Credenciais inválidas.' }); } const token = signToken({ id: user.id, username: user.username, role: user.role, email: user.email }); return res.json({ ok: true, token, user }); }); app.get('/api/dashboard', requireAuth, (req, res) => { res.json({ ok: true, data: getDashboardData() }); }); app.get('/api/settings', requireAuth, (req, res) => { const rows = db.prepare('SELECT key, value, updated_at FROM settings ORDER BY key').all(); const settings = Object.fromEntries(rows.map((item) => [item.key, item.value])); res.json({ ok: true, settings }); }); app.post('/api/settings', requireAuth, (req, res) => { const updates = req.body || {}; const stmt = db.prepare(`INSERT INTO settings (key, value, updated_at) VALUES (?, ?, ?) ON CONFLICT(key) DO UPDATE SET value=excluded.value, updated_at=CURRENT_TIMESTAMP`); for (const [key, value] of Object.entries(updates)) { stmt.run(key, String(value), new Date().toISOString()); } logAudit(req.user.username, 'update_settings', 'settings', 'global', JSON.stringify(updates)); res.json({ ok: true, message: 'Configurações salvas com sucesso.' }); }); app.get('/api/panels', requireAuth, (req, res) => { const panels = db.prepare('SELECT * FROM panels ORDER BY created_at DESC').all(); res.json({ ok: true, panels }); }); app.post('/api/panels', requireAuth, (req, res) => { const { name, title, description, channel_id, category_id, embed_json, welcome_message, status } = req.body || {}; const stmt = db.prepare(` INSERT INTO panels (name, title, description, channel_id, category_id, embed_json, welcome_message, status) VALUES (?, ?, ?, ?, ?, ?, ?, ?) `); const result = stmt.run(name, title, description || '', channel_id || '', category_id || '', JSON.stringify(embed_json || {}), welcome_message || '', status || 'draft'); logAudit(req.user.username, 'create_panel', 'panel', String(result.lastInsertRowid), JSON.stringify({ name, title })); res.status(201).json({ ok: true, id: result.lastInsertRowid, message: 'Painel criado.' }); }); app.get('/api/tickets', requireAuth, (req, res) => { const tickets = db.prepare('SELECT * FROM tickets ORDER BY created_at DESC').all(); res.json({ ok: true, tickets }); }); app.post('/api/tickets', requireAuth, (req, res) => { const { user_id, user_name, category, reason, priority = 'normal', panel_id = 1 } = req.body || {}; if (!user_id || !reason) { return res.status(400).json({ ok: false, message: 'Usuário e motivo são obrigatórios.' }); } const existing = db.prepare('SELECT id FROM tickets WHERE user_id = ? AND status IN (\'open\', \'in_progress\', \'pending\') ORDER BY created_at DESC LIMIT 1').get(user_id); if (existing) { return res.status(409).json({ ok: false, message: 'Usuário já possui um ticket ativo.' }); } const stmt = db.prepare(` INSERT INTO tickets (user_id, user_name, category, reason, status, priority, opened_by, panel_id, metadata) VALUES (?, ?, ?, ?, 'open', ?, ?, ?, ?) `); const result = stmt.run(user_id, user_name || 'Usuário', category || 'Suporte geral', reason, priority, req.user.username, panel_id, JSON.stringify({ created_from: 'web' })); logAudit(req.user.username, 'create_ticket', 'ticket', String(result.lastInsertRowid), JSON.stringify({ user_id, category, priority })); res.status(201).json({ ok: true, ticketId: result.lastInsertRowid, message: 'Ticket criado com sucesso.' }); }); app.post('/api/tickets/:id/assign', requireAuth, (req, res) => { const { assignee_id, assignee_name } = req.body || {}; const id = Number(req.params.id); const ticket = db.prepare('SELECT * FROM tickets WHERE id = ?').get(id); if (!ticket) return res.status(404).json({ ok: false, message: 'Ticket não encontrado.' }); db.prepare(`UPDATE tickets SET assignee_id = ?, assignee_name = ?, status = 'in_progress', last_sync_at = ? WHERE id = ?`).run(assignee_id || req.user.id, assignee_name || req.user.username, new Date().toISOString(), id); logAudit(req.user.username, 'assign_ticket', 'ticket', String(id), JSON.stringify({ assignee_id, assignee_name })); res.json({ ok: true, message: 'Ticket atribuído com sucesso.' }); }); app.post('/api/tickets/:id/close', requireAuth, (req, res) => { const { close_reason = 'Encerrado pelo painel administrativo' } = req.body || {}; const id = Number(req.params.id); db.prepare(`UPDATE tickets SET status = 'closed', closed_at = ?, close_reason = ?, last_sync_at = ? WHERE id = ?`).run(new Date().toISOString(), close_reason, new Date().toISOString(), id); logAudit(req.user.username, 'close_ticket', 'ticket', String(id), close_reason); res.json({ ok: true, message: 'Ticket encerrado.' }); }); app.post('/api/tickets/:id/reopen', requireAuth, (req, res) => { const id = Number(req.params.id); db.prepare(`UPDATE tickets SET status='open', closed_at = NULL, close_reason = NULL, last_sync_at = ? WHERE id = ?`).run(new Date().toISOString(), id); logAudit(req.user.username, 'reopen_ticket', 'ticket', String(id), 'Ticket reaberto'); res.json({ ok: true, message: 'Ticket reaberto.' }); }); app.get('/api/transcripts/:ticketId', requireAuth, (req, res) => { const transcript = db.prepare('SELECT * FROM transcripts WHERE ticket_id = ? ORDER BY created_at DESC LIMIT 1').get(Number(req.params.ticketId)); if (!transcript) { return res.status(404).json({ ok: false, message: 'Transcrição não encontrada.' }); } res.json({ ok: true, transcript }); }); app.post('/api/transcripts/:ticketId', requireAuth, (req, res) => { const ticketId = Number(req.params.ticketId); const content = req.body?.content || 'Transcrição disponível.'; db.prepare(`INSERT INTO transcripts (ticket_id, content, type) VALUES (?, ?, 'txt') ON CONFLICT(ticket_id, type) DO UPDATE SET content=excluded.content, created_at=CURRENT_TIMESTAMP`).run(ticketId, content); logAudit(req.user.username, 'write_transcript', 'ticket', String(ticketId), 'Transcrição salva'); res.json({ ok: true, message: 'Transcrição registrada.' }); }); app.get('/api/logs', requireAuth, (req, res) => { const logs = db.prepare('SELECT * FROM audit_logs ORDER BY created_at DESC LIMIT 50').all(); res.json({ ok: true, logs }); }); app.post('/api/bot/sync', (req, res) => { const apiKey = req.headers['x-bot-api-key'] || req.body?.apiKey || ''; if (!apiKey || apiKey !== config.botApiKey) { return res.status(401).json({ ok: false, message: 'Chave de autenticação do bot inválida.' }); } const { status = 'online', source = 'discord', lastSync = new Date().toISOString(), error = null } = req.body || {}; const stmt = db.prepare(`INSERT INTO bot_status (is_online, last_sync, uptime_seconds, error) VALUES (?, ?, ?, ?)`); stmt.run(status === 'online' ? 1 : 0, lastSync, Number(process.uptime()), error || null); logAudit('bot', 'sync_status', 'bot', 'status', JSON.stringify({ source, status, lastSync, error })); res.json({ ok: true, message: 'Sincronização recebida com sucesso.', status, lastSync, source, }); }); app.get('/api/bot/status', requireAuth, (req, res) => { const row = db.prepare('SELECT * FROM bot_status ORDER BY id DESC LIMIT 1').get(); res.json({ ok: true, bot: row || { is_online: 0, last_sync: null, error: 'Sem status registrado' } }); }); app.get('*', (req, res) => { res.sendFile(path.join(publicDir, 'index.html')); }); ensureDefaultAdmin(); app.listen(config.port, () => { console.log(`${config.appName} ouvindo na porta ${config.port}`); });
